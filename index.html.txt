// LOGIN

function login(){


let u=document.getElementById("username").value;

let p=document.getElementById("password").value;



if(u==="Adm_123" && p==="Geo_123"){

document.getElementById("login").style.display="none";

document.getElementById("admin").classList.remove("hidden");

displayBooks();

}



else if(u==="User_123" && p==="User_geo"){


document.getElementById("login").style.display="none";

document.getElementById("user").classList.remove("hidden");

displayUserBooks();


}


else{

document.getElementById("message").innerHTML="Invalid Login";

}



}



// ADD BOOK

function addBook(){


let book={

title:title.value,
author:author.value,
isbn:isbn.value,
category:category.value,
copies:copies.value

};


books.push(book);


alert("Book Added");

displayBooks();


}



// DISPLAY ADMIN BOOKS

function displayBooks(){


let table="";


let search=document.getElementById("search")?.value.toLowerCase() || "";


books.filter(b=>

b.title.toLowerCase().includes(search)

).forEach((b,i)=>{


table+=`

<tr>

<td>${b.title}</td>

<td>${b.author}</td>

<td>${b.isbn}</td>

<td>${b.category}</td>

<td>

<button onclick="deleteBook(${i})">
Delete
</button>

</td>


</tr>

`;


});


document.getElementById("adminBooks").innerHTML=table;


}




function deleteBook(i){

books.splice(i,1);

displayBooks();

}




// USER DISPLAY


function displayUserBooks(){


let table="";


let search=document.getElementById("userSearch")?.value.toLowerCase() || "";



books.filter(b=>

b.title.toLowerCase().includes(search)

).forEach(b=>{


// hide ERF books

if(b.category!="ERF Books"){


table+=`

<tr>

<td>${b.title}</td>

<td>${b.author}</td>

<td>${b.isbn}</td>

<td>${b.category}</td>

</tr>


`;

}


});


document.getElementById("userBooks").innerHTML=table;


}




// CSV EXPORT


function exportCSV(){


let csv="Title,Author,ISBN,Category,Copies\n";


books.forEach(b=>{


csv+=`${b.title},${b.author},${b.isbn},${b.category},${b.copies}\n`;


});


let blob=new Blob([csv],{type:"text/csv"});


let link=document.createElement("a");

link.href=URL.createObjectURL(blob);

link.download="Geography_Library_Books.csv";

link.click();


}
